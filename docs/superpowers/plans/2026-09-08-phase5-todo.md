"""待办 ORM 模型。"""
from sqlalchemy import Column, String, Text, JSON, DateTime, ForeignKey
from sqlalchemy.dialects.postgresql import UUID
from app.db.base import Base
import uuid

class Todo(Base):
    """待办任务。"""
    __tablename__ = "todos"

    id = Column(UUID(as_uuid=True), primary_key=True, default=uuid.uuid4)
    title = Column(String(255), nullable=False)
    source = Column(Text, comment="来源描述")
    status = Column(String(20), default="pending", comment="pending/done/rejected")
    deadline = Column(Integer, comment="截止时间（秒）")
    form_schema = Column(JSON, default=list, comment="表单字段定义")
    form_data = Column(JSON, default=dict, comment="提交的表单数据")
    submitted_at = Column(DateTime(timezone=True))
    execution_id = Column(UUID(as_uuid=True), ForeignKey("workflow_executions.id"), comment="关联的工作流执行")
    node_id = Column(String(100), comment="人工节点ID")
    user_id = Column(UUID(as_uuid=True), ForeignKey("users.id"))
    created_at = Column(DateTime(timezone=True), server_default="now()")
```

"""待办路由：CRUD + 提交。"""
from fastapi import APIRouter, Depends
from sqlalchemy import select
from datetime import datetime
from app.api.deps import get_current_user
from app.api.response import ok
from app.db.session import async_session
from app.models.todo import Todo
from app.models.user import User
from app.schemas.todo import TodoOut, TodoSubmit
from app.services.workflow_service import WorkflowService

router = APIRouter(tags=["todos"])

@router.get("/todos")
async def list_todos(me: User = Depends(get_current_user)):
    """列出待办任务。"""
    async with async_session() as s:
        rows = (await s.execute(
            select(Todo)
            .where(Todo.user_id == me.id)
            .where(Todo.status == "pending")
            .order_by(Todo.created_at.desc())
        )).scalars().all()
    return ok([TodoOut(...).model_dump() for t in rows])

@router.put("/todos/{todo_id}")
async def update_todo(
    todo_id: str,
    body: dict,
    me: User = Depends(get_current_user)
):
    """更新待办。"""
    async with async_session() as s:
        todo = (await s.execute(
            select(Todo).where(Todo.id == todo_id)
        )).scalar_one_or_none()

        if not todo:
            raise BizException(ErrorCode.NOT_FOUND, "待办不存在")

        # 更新字段
        for key, value in body.items():
            if hasattr(todo, key):
                setattr(todo, key, value)

        await s.commit()
        await s.refresh(todo)

    return ok(TodoOut(...).model_dump())

@router.post("/todos/{todo_id}/submit")
async def submit_todo(
    todo_id: str,
    body: TodoSubmit,
    me: User = Depends(get_current_user)
):
    """提交待办。"""
    async with async_session() as s:
        todo = (await s.execute(
            select(Todo).where(Todo.id == todo_id)
        )).scalar_one_or_none()

        if not todo:
            raise BizException(ErrorCode.NOT_FOUND, "待办不存在")

        if todo.status != "pending":
            raise BizException(ErrorCode.PARAM_ERROR, "待办已处理")

        # 更新状态
        todo.status = "done"
        todo.form_data = body.form_data
        todo.submitted_at = datetime.utcnow()

        await s.commit()

        # 恢复工作流执行
        if todo.execution_id:
            wf_service = WorkflowService()
            await wf_service.resume_execution(
                str(todo.execution_id),
                todo.node_id,
                body.form_data
            )

    return ok(TodoOut(...).model_dump())
```

"""人工节点执行器。"""
from app.core.workflow_engine.state import WorkflowState
from app.models.todo import Todo
from app.db.session import async_session

async def execute_human_node(state: WorkflowState, config: dict) -> WorkflowState:
    """执行人工节点。

    创建待办任务并暂停工作流执行。
    """
    # 从配置中提取表单定义
    form_schema = config.get("form_schema", [])
    title = config.get("title", "人工审核")

    # 创建待办任务
    async with async_session() as s:
        todo = Todo(
            title=title,
            source=f"工作流 {state.get('workflow_name', '')}",
            status="pending",
            form_schema=form_schema,
            execution_id=state.get("execution_id"),
            node_id=state.get("current_node"),
            user_id=state.get("user_id")
        )
        s.add(todo)
        await s.commit()

    # 标记工作流为等待状态
    return {
        **state,
        "status": "wait",
        "todo_id": str(todo.id)
    }
```

async def resume_execution(
    self,
    execution_id: str,
    node_id: str,
    form_data: dict
):
    """恢复工作流执行。

    Args:
        execution_id: 执行实例ID
        node_id: 人工节点ID
        form_data: 表单数据
    """
    async with async_session() as s:
        execution = (await s.execute(
            select(WorkflowExecution).where(WorkflowExecution.id == execution_id)
        )).scalar_one_or_none()

        if not execution:
            return

        # 更新节点状态
        node_states = execution.node_states or {}
        node_states[node_id] = {
            "status": "done",

"""待办+工作流集成测试。"""
import pytest
from httpx import AsyncClient

@pytest.mark.asyncio
async def test_human_workflow_integration(async_client: AsyncClient, auth_headers: dict):
    """测试人工审核工作流完整流程。"""
    # 1. 创建包含人工节点的工作流
    wf_resp = await async_client.post(
        "/api/v2/workflows",
        json={
            "name": "审批流程",
            "nodes": [
                {"id": "start", "type": "start"},
                {"id": "human_1", "type": "human", "config": {"title": "审批"}},
                {"id": "end", "type": "end"}
            ]
        },
        headers=auth_headers
    )
    wf_id = wf_resp.json()["data"]["id"]

    # 2. 执行工作流
    exec_resp = await async_client.post(
        f"/api/v2/workflows/{wf_id}/execute",
        json={},
        headers=auth_headers
    )
    execution_id = exec_resp.json()["data"]["executionId"]

    # 3. 等待工作流到达人工节点
    import asyncio
    await asyncio.sleep(2)

    # 4. 获取待办列表
    todos_resp = await async_client.get("/api/v2/todos", headers=auth_headers)
    assert len(todos_resp.json()["data"]) > 0

    todo_id = todos_resp.json()["data"][0]["id"]

    # 5. 提交待办
    submit_resp = await async_client.post(
        f"/api/v2/todos/{todo_id}/submit",
        json={"form_data": {"approved": True}},
        headers=auth_headers
    )
    assert submit_resp.json()["data"]["status"] == "done"

    # 6. 验证工作流继续执行
    await asyncio.sleep(2)

    exec_detail = await async_client.get(
        f"/api/v2/executions/{execution_id}",
        headers=auth_headers
    )
    assert exec_detail.json()["data"]["status"] == "success"
```
